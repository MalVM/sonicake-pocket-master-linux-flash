#!/usr/bin/env python3
"""HTFW container parser.

Layout taken from 5868USB.dll sub_180092eb0 (header reader) and sub_180092640
(isRealFirmware: validates magic + model only). Verified against stock
Firmware V1.3.3 (2,056,820 B) and PocketMaster_CFW0.9.8_base1.3.3 (2,125,752 B):
in both, field 0x23 == file_size - 136 and an 80-byte all-0xFF trailer follows.
"""
import struct
import hashlib
import json
import sys

MAGIC = b"HTFW"
HDR_LEN = 0x38
TRAILER_LEN = 80


def parse(path):
    with open(path, "rb") as fh:
        b = fh.read()
    if len(b) < HDR_LEN or b[:4] != MAGIC:
        raise ValueError("%s: not an HTFW container" % path)

    def u32(o):
        return struct.unpack_from("<I", b, o)[0]

    def u16(o):
        return struct.unpack_from("<H", b, o)[0]

    f = {
        "magic": b[0:4].decode("ascii", "replace"),
        "u04": u16(0x04),          # differs per image; nothing in the DLL compares it
        "u06": u16(0x06),          # 1 in every image seen
        "image_size": u32(0x08),   # == file length in both known images
        "model": b[0x0C:0x1C].split(b"\0")[0].decode("ascii", "replace"),
        "b1C": b[0x1C],            # 0x56 in every image seen
        "version": "%d.%d.%d" % (b[0x1D], b[0x1E], b[0x1F]),
        "b20_22": list(b[0x20:0x23]),
        "payload_len": u32(0x23),  # file_size - 136 in both known images
        "b27": b[0x27],
        "digest_slot": b[0x28:0x38].hex(),
        "file_size": len(b),
    }
    f["payload"] = b[HDR_LEN:HDR_LEN + f["payload_len"]]
    f["trailer"] = b[HDR_LEN + f["payload_len"]:]
    f["payload_sha256"] = hashlib.sha256(f["payload"]).hexdigest()

    w = f["warnings"] = []
    if f["image_size"] != f["file_size"]:
        w.append("0x08 (%d) != file length (%d)" % (f["image_size"], f["file_size"]))
    if len(f["trailer"]) != TRAILER_LEN:
        w.append("trailer is %d B, expected %d" % (len(f["trailer"]), TRAILER_LEN))
    elif f["trailer"].count(0xFF) != TRAILER_LEN:
        w.append("trailer is not all 0xFF")
    if f["u06"] != 1:
        w.append("u06 is %d, expected 1" % f["u06"])
    if f["b1C"] != 0x56:
        w.append("0x1C is %#x, expected 0x56" % f["b1C"])
    f["sane"] = not w
    return f


def check_model(f, expected):
    """Mirror isRealFirmware: the DLL's tag is literally 'equalsIgnoreCase'."""
    return f["model"].lower() == expected.lower()


def main(argv):
    if not argv:
        print("usage: htfw.py FILE.bin [FILE.bin ...]", file=sys.stderr)
        return 2
    rc = 0
    for p in argv:
        try:
            f = parse(p)
        except ValueError as e:
            print("%s\n  ERROR: %s" % (p, e), file=sys.stderr)
            rc = 1
            continue
        shown = dict((k, v) for k, v in f.items() if k not in ("payload", "trailer"))
        print(p)
        print(json.dumps(shown, indent=2))
        for w in f["warnings"]:
            print("  WARNING: %s" % w, file=sys.stderr)
            rc = 1
    return rc


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))

cat > sxproto.py <<'PYEOF'
#!/usr/bin/env python3
"""5868USB SysEx codec, recovered from 5868USB.dll.

  sub_180095210  encoder: [crc8][count][index][type] ++ payload, nibble-expanded
  sub_18008fca0  receiver: same, verified with the same CRC table
  sub_180095010  chunking: 42 B if type==0 else 19 B; terminator is F0 F7

Importing this module self-checks the CRC table against the constants read out
of the DLL, so a wrong table fails loudly instead of silently.
"""
import sys
import binascii


def _table():
    t = []
    for i in range(256):
        c = i
        for _ in range(8):
            c = ((c << 1) ^ 0x07) & 0xFF if c & 0x80 else (c << 1) & 0xFF
        t.append(c)
    return bytes(t)


CRC8_TABLE = _table()
# constants read directly from the table at 0x18013a400 in 5868USB.dll
assert CRC8_TABLE[0] == 0x00 and CRC8_TABLE[1] == 0x07 and CRC8_TABLE[2] == 0x0E \
    and CRC8_TABLE[3] == 0x09 and CRC8_TABLE[8] == 0x38 and CRC8_TABLE[9] == 0x3F, \
    "CRC-8 table does not match 5868USB.dll"


def crc8(data, crc=0):
    for b in data:
        crc = CRC8_TABLE[crc ^ b]
    return crc


def expand(b):
    """Each byte -> two wire bytes, high nibble first."""
    out = bytearray()
    for x in b:
        out.append(x >> 4)
        out.append(x & 0x0F)
    return bytes(out)


def collapse(b):
    return bytes((b[i] << 4) | b[i + 1] for i in range(0, len(b) - 1, 2))


def frame(count, index, type_byte, payload=b""):
    body = bytearray([0, count & 0xFF, index & 0xFF, type_byte & 0xFF])
    body += bytearray(payload)
    body[0] = crc8(body[1:])
    return b"\xF0" + expand(bytes(body)) + b"\xF7"


def chunk_size(type_byte):
    """cmove at 0x180095061: 0x2A when the type byte is zero, else 0x13."""
    return 42 if type_byte == 0 else 19


def terminator():
    """What sub_180095010 returns at end of transfer: a bare F0 F7."""
    return b"\xF0\xF7"


def parse(msg):
    if not msg or msg[0] != 0xF0 or msg[-1] != 0xF7:
        return None
    body = collapse(msg[1:-1])
    if len(body) < 4:
        return None
    return {
        "crc_ok": crc8(body[1:]) == body[0],
        "crc": body[0],
        "crc_calc": crc8(body[1:]),
        "count": body[1],
        "index": body[2],
        "type": body[3],
        "payload": body[4:],
        "len": len(body) - 4,
    }


def main(argv):
    if not argv:
        print("usage: sxproto.py HEXFRAME [HEXFRAME ...]", file=sys.stderr)
        print("       echo f0....f7 | sxproto.py -", file=sys.stderr)
        return 2
    lines = sys.stdin.read().splitlines() if argv == ["-"] else argv
    for line in lines:
        line = line.strip()
        if not line:
            continue
        try:
            p = parse(bytes.fromhex(line))
        except ValueError:
            print("not hex: %s" % line[:64])
            continue
        if p is None:
            print("not a framed SysEx: %s" % line[:64])
            continue
        print("crc=%s count=%3d index=%3d type=%#04x len=%3d %s" % (
            "ok " if p["crc_ok"] else "BAD", p["count"], p["index"],
            p["type"], p["len"], binascii.hexlify(p["payload"][:24]).decode()))
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))

#!/usr/bin/env python3
"""Dump USB interfaces/endpoints. Read-only: claims nothing, sends nothing.

Run it once with the pedal in normal mode and once in flash mode. The delta is
the answer to 'is flash mode USB-MIDI class (01/03) or vendor-specific (ff)?'
  class 01 subclass 03 -> use the ALSA transport, libusb is wrong (snd-usb-audio owns it)
  class ff             -> use the libusb transport
"""
import sys

try:
    import usb.core
    import usb.util
except ImportError:
    sys.exit("needs pyusb:  pip install --user pyusb")

CLASS = {0x01: "AUDIO", 0x03: "HID", 0x08: "MASS-STORAGE", 0xFF: "VENDOR"}
SUB_AUDIO = {0x01: "AudioControl", 0x02: "AudioStreaming", 0x03: "MIDIStreaming"}


def show(dev):
    try:
        print("\n%04x:%04x  bus %s addr %s  bcdUSB %04x  bDeviceClass %02x"
              % (dev.idVendor, dev.idProduct, dev.bus, dev.address,
                 dev.bcdUSB, dev.bDeviceClass))
        for cfg in dev:
            print("  config %s" % cfg.bConfigurationValue)
            for intf in cfg:
                cn = CLASS.get(intf.bInterfaceClass, hex(intf.bInterfaceClass))
                sub = ""
                if intf.bInterfaceClass == 1:
                    sub = SUB_AUDIO.get(intf.bInterfaceSubClass, "")
                print("    intf %d.%d  class %02x %s %s  subclass %02x  proto %02x"
                      % (intf.bInterfaceNumber, intf.bAlternateSetting,
                         intf.bInterfaceClass, cn, sub,
                         intf.bInterfaceSubClass, intf.bInterfaceProtocol))
                for ep in intf:
                    print("      ep %02x %s %s maxpkt %d interval %d"
                          % (ep.bEndpointAddress,
                             "IN" if ep.bEndpointAddress & 0x80 else "OUT",
                             usb.util.endpoint_type(ep.bmAttributes),
                             ep.wMaxPacketSize, ep.bInterval))
    except usb.core.USBError as e:
        print("  [cannot read descriptors: %s]" % e)


def main(argv):
    want = set(int(x, 0) for x in argv)
    devs = list(usb.core.find(find_all=True))
    if not devs:
        print("no USB devices visible (permissions? try as root once)")
        return 1
    for dev in devs:
        if want and dev.idVendor not in want:
            continue
        show(dev)
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))

#!/usr/bin/env python3
"""Upload a firmware image to a 5868USB device using the recovered protocol.

DANGEROUS BY DESIGN GUARD: dry run unless --i-understand-the-risk is given.
The frame format, CRC, chunking and state machine are recovered from the DLL.
Two things are NOT known: which type byte marks a firmware image, and what the
device replies with (there is no ACK handling here - this paces blindly).
"""
import argparse
import sys
import time

import htfw
import sxproto


def announce(payload, type_byte, entry_id, flags):
    """sub_180092b70: 11 61 <flags> 00 00 00 <u32 entry+0x18> <u32 size>."""
    ann = bytes([0x11, 0x61, flags & 0xFF, 0, 0, 0])
    ann += entry_id.to_bytes(4, "little")
    ann += len(payload).to_bytes(4, "little")
    return sxproto.frame(1, 0, type_byte, ann)


def sequence(payload, type_byte, entry_id=0, flags=0, block=255):
    seq = [announce(payload, type_byte, entry_id, flags)]
    cs = sxproto.chunk_size(type_byte)
    idx = 0
    for off in range(0, len(payload), cs):
        if idx == block:
            idx = 0
        seq.append(sxproto.frame(0xFF, idx, type_byte, payload[off:off + cs]))
        idx += 1
    seq.append(sxproto.terminator())
    seq.append(sxproto.frame(1, 0, type_byte, b"\x6e\x00"))  # sub_180091a00
    return seq


def usb_packetise(sysex):
    """SysEx -> USB-MIDI 1.0 4-byte event packets."""
    out = bytearray()
    i = 0
    while i < len(sysex):
        c = sysex[i:i + 3]
        cin = 0x04 if len(c) == 3 else (0x06 if len(c) == 2 else 0x05)
        out += bytes([cin]) + c + b"\0" * (3 - len(c))
        i += 3
    return bytes(out)


def open_alsa(port):
    import rtmidi
    out = rtmidi.MidiOut()
    ports = out.get_ports()
    if isinstance(port, int):
        out.open_port(port)
    else:
        if port not in ports:
            sys.exit("port %r not found; available: %r" % (port, ports))
        out.open_port(ports.index(port))
    return lambda msg: out.send_message(list(msg))


def open_usb(vid, pid):
    import usb.core
    import usb.util
    dev = usb.core.find(idVendor=vid, idProduct=pid)
    if dev is None:
        sys.exit("device %04x:%04x not found" % (vid, pid))
    intf = ep = None
    for cfg in dev:
        for i in cfg:
            if i.bInterfaceClass == 0xFF:
                intf = i
                ep = next(e for e in i if not (e.bEndpointAddress & 0x80))
    if intf is None:
        sys.exit("no vendor-specific interface; the device is probably MIDI class "
                 "-> use --transport alsa instead")
    try:
        dev.detach_kernel_driver(intf.bInterfaceNumber)
    except Exception:
        pass
    usb.util.claim_interface(dev, intf.bInterfaceNumber)
    return lambda msg: dev.write(ep.bEndpointAddress, usb_packetise(msg), timeout=2000)


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("firmware")
    ap.add_argument("--type", type=lambda s: int(s, 0), default=0,
                    help="type byte (UNKNOWN for firmware; 0 selects 42-byte chunks)")
    ap.add_argument("--entry-id", type=lambda s: int(s, 0), default=0,
                    help="u32 at entry+0x18 (UNKNOWN meaning)")
    ap.add_argument("--flags", type=lambda s: int(s, 0), default=0)
    ap.add_argument("--transport", choices=["none", "alsa", "usb"], default="none")
    ap.add_argument("--port", default=None)
    ap.add_argument("--vid", type=lambda s: int(s, 0), default=None)
    ap.add_argument("--pid", type=lambda s: int(s, 0), default=None)
    ap.add_argument("--pace", type=float, default=0.002)
    ap.add_argument("--confirm-model", default=None)
    ap.add_argument("--i-understand-the-risk", action="store_true")
    a = ap.parse_args()

    raw = open(a.firmware, "rb").read()
    if raw[:4] == b"HTFW":
        fw = htfw.parse(a.firmware)
        print("HTFW: model %r, container claims v%s, %d B payload (%d B trailer dropped)"
              % (fw["model"], fw["version"], len(fw["payload"]), len(fw["trailer"])),
              file=sys.stderr)
        for w in fw["warnings"]:
            print("  WARNING: %s" % w, file=sys.stderr)
        if a.confirm_model and not htfw.check_model(fw, a.confirm_model):
            sys.exit("refusing: model %r != %r" % (fw["model"], a.confirm_model))
        payload = fw["payload"]
    else:
        if a.confirm_model:
            print("note: bare image, model check not possible", file=sys.stderr)
        payload = raw

    seq = sequence(payload, a.type, a.entry_id, a.flags)
    print("%d frames, %d wire bytes, %d B payload, %d B chunks"
          % (len(seq), sum(len(f) for f in seq), len(payload),
             sxproto.chunk_size(a.type)))

    if a.transport == "none" or not a.i_understand_the_risk:
        for i, f in enumerate(seq[:5]):
            print("  %4d %3dB %s" % (i, len(f), f.hex()))
        print("  ...")
        print("  %4d %3dB %s" % (len(seq) - 1, len(seq[-1]), seq[-1].hex()))
        print("dry run: nothing sent")
        return 0

    if a.transport == "alsa":
        send = open_alsa(a.port if isinstance(a.port, int) else a.port)
    else:
        if a.vid is None or a.pid is None:
            sys.exit("--vid and --pid are required for the usb transport")
        send = open_usb(a.vid, a.pid)

    t0 = time.time()
    for i, f in enumerate(seq):
        send(f)
        if a.pace:
            time.sleep(a.pace)
        if i % 500 == 0:
            print("  %d/%d (%.1fs)" % (i, len(seq), time.time() - t0), file=sys.stderr)
    print("done in %.1fs" % (time.time() - t0))
    return 0


if __name__ == "__main__":
    sys.exit(main())
